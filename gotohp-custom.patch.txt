+// Upstream hard-codes a single Pixel profile, so every installation of the

+// build, or a single run, can present a coherent and unique client profile.

+// Build-time overrides (go build -ldflags):
+//
+//	go build -tags cli -ldflags "-X app/core.DefaultDeviceModel=Pixel" -o gotohp-cli.exe .
+//
+// Runtime overrides (environment variables):
+//
+//	GOTOHP_DEVICE_MODEL      device model, e.g. "Pixel" or "Pixel XL"
+//	GOTOHP_DEVICE_MAKE       device manufacturer
+//	GOTOHP_ANDROID_VERSION   human readable Android version in the UA
+//	GOTOHP_ANDROID_API       numeric SDK level in the protobuf
+//	GOTOHP_BUILD             build fingerprint in the UA
+//	GOTOHP_CLIENT_VERSION    Google Photos client version code
+//	GOTOHP_CRONET_VERSION    Cronet build in the UA

+	DefaultDeviceMake = "Google"

+	DefaultDeviceAndroidVer = "Android 9"

+// parseKeyValues reads a simple key=value file.
+func parseKeyValues(text string) map[string]string {
+	out := map[string]string{}
+	for _, line := range strings.Split(text, "\n") {
+		line = strings.TrimSpace(line)
+		if line == "" || strings.HasPrefix(line, "#") {
+			continue
+		}
+		k, v, ok := strings.Cut(line, "=")
+		if !ok {
+			continue
+		}
+		out[strings.TrimSpace(k)] = strings.TrimSpace(v)
+	}
+	return out
+}

+			return 0
+		}
+		n = n*10 + int64(c-'0')
+	}
+	return n

+// dialThroughProxy opens a tunnel to targetAddr through an HTTP proxy.
+func dialThroughProxy(ctx context.Context, proxyURL *url.URL, targetAddr string, timeout time.Duration) (net.Conn, error) {
+	proxyAddr := proxyURL.Host
+	if proxyAddr == "" {
+		proxyAddr = proxyURL.Opaque
+	}
+	if proxyURL.Port() == "" {
+		switch strings.ToLower(proxyURL.Scheme) {
+		case "https":
+			proxyAddr = net.JoinHostPort(proxyURL.Hostname(), "443")
+		default:
+			proxyAddr = net.JoinHostPort(proxyURL.Hostname(), "80")
+		}
+	}
+
+	dialer := &net.Dialer{Timeout: timeout}
+	conn, err := dialer.DialContext(ctx, "tcp", proxyAddr)
+	if err != nil {
+		return nil, err
+	}
+
+	req := &http.Request{
+		Method: http.MethodConnect,
+		URL:    &url.URL{Opaque: targetAddr},
+		Host:   targetAddr,
+		Header: http.Header{},
+	}
+	if user := proxyURL.User; user != nil {
+		password, _ := user.Password()
+		req.Header.Set("Proxy-Authorization", "Basic "+basicAuth(user.Username(), password))
+	}
+
+	if deadline, ok := ctx.Deadline(); ok {
+		_ = conn.SetDeadline(deadline)
+	} else {
+		_ = conn.SetDeadline(time.Now().Add(timeout))
+	}
+
+	if err := req.Write(conn); err != nil {
+		conn.Close()
+		return nil, err
+	}
+
+	br := bufio.NewReader(conn)
+	resp, err := http.ReadResponse(br, req)
+	if err != nil {
+		conn.Close()
+		return nil, err
+	}
+	resp.Body.Close()
+	if resp.StatusCode != http.StatusOK {
+		conn.Close()
+		return nil, fmt.Errorf("proxy CONNECT failed: %s", resp.Status)
+	}
+	_ = conn.SetDeadline(time.Time{})
+
+	return &bufferedConn{Conn: conn, reader: br}, nil
+}

+// basicAuth builds a base64 encoded basic auth credential.
+func basicAuth(user, password string) string {
+	return base64.StdEncoding.EncodeToString([]byte(user + ":" + password))

+// newFingerprintDialer returns a dialer that performs a uTLS handshake with
+// the selected browser profile, optionally tunnelling through an HTTP proxy.
+func newFingerprintDialer(profile string, proxyURL *url.URL, timeout time.Duration) func(ctx context.Context, network, addr string) (net.Conn, error) {
+	return func(ctx context.Context, network, addr string) (net.Conn, error) {
+		host, _, err := net.SplitHostPort(addr)
+		if err != nil {
+			return nil, err
+		}
+
+		helloID, ok := chromeHelloID(profile)
+		if !ok {
+			return nil, fmt.Errorf("unsupported TLS fingerprint profile: %s", profile)
+		}
+
+		var rawConn net.Conn
+		if proxyURL != nil {
+			rawConn, err = dialThroughProxy(ctx, proxyURL, addr, timeout)
+		} else {
+			dialer := &net.Dialer{Timeout: timeout}
+			rawConn, err = dialer.DialContext(ctx, network, addr)
+		}
+		if err != nil {
+			return nil, err
+		}
+
+		spec, err := utls.UTLSIdToSpec(helloID)
+		if err != nil {
+			rawConn.Close()
+			return nil, err
+		}
+		// Speak only HTTP/1.1 in ALPN: the connection returned here is not a
+		// *tls.Conn, so the standard library cannot negotiate HTTP/2 on it.
+		for i := range spec.Extensions {
+			if alpn, ok := spec.Extensions[i].(*utls.ALPNExtension); ok {
+				alpn.AlpnProtocols = []string{"http/1.1"}
+			}
+		}
+
+		uconn := utls.UClient(rawConn, &utls.Config{
+			ServerName: host,
+			MinVersion: tls.VersionTLS12,
+		}, utls.HelloCustom)
+
+		if err := uconn.ApplyPreset(&spec); err != nil {
+			rawConn.Close()
+			return nil, err
+		}
+		if err := uconn.HandshakeContext(ctx); err != nil {
+			rawConn.Close()
+			return nil, err
+		}
+		state := uconn.ConnectionState()
+		if !state.HandshakeComplete || len(state.PeerCertificates) == 0 {
+			rawConn.Close()
+			return nil, fmt.Errorf("TLS handshake did not complete")
+		}
+		return uconn, nil
+	}
+}

+// applyTLSFingerprint installs the fingerprint dialer on the transport when
+// emulation is enabled. The standard TLSClientConfig is kept so certificate
+// verification still happens, and any failure falls back to the stock stack.
+func applyTLSFingerprint(transport *http.Transport, proxyURL *url.URL, timeout time.Duration) {
+	if !tlsFingerprintEnabled() {
+		return
+	}
+	profile := TLSFingerprintName()
+	dialTLS := newFingerprintDialer(profile, proxyURL, timeout)
+
+	transport.DialTLSContext = func(ctx context.Context, network, addr string) (net.Conn, error) {
+		conn, err := dialTLS(ctx, network, addr)
+		if err == nil {
+			return conn, nil
+		}
+		// Fall back to the standard Go TLS stack so uploads keep working.
+		d := &tls.Dialer{Config: transport.TLSClientConfig, NetDialer: &net.Dialer{Timeout: timeout}}
+		return d.DialContext(ctx, network, addr)
+	}

+	fmt.Printf("  device model    : %s\n", core.IdentityModel())
+	fmt.Printf("  device make     : %s\n", core.IdentityMake())
+	fmt.Printf("  android version : %s (API %d)\n", core.IdentityAndroidVer(), core.IdentityAndroidAPI())
+	fmt.Printf("  build           : %s\n", core.IdentityBuild())
+	fmt.Printf("  client version  : %d\n", core.IdentityClientVersionCode())
+	fmt.Printf("  cronet          : %s\n", core.IdentityCronetVersion())
+	fmt.Printf("  language        : %s\n", core.IdentityLanguage(""))
+	fmt.Printf("  app user-agent  : %s\n", core.AppUserAgent(core.IdentityLanguage("")))
+	fmt.Printf("  googleauth ua   : %s\n", core.IdentityAuthUserAgent())
+	fmt.Printf("  tls fingerprint : %s\n", core.TLSFingerprintName())
+	return nil
+}

+func generateIdentity(cmd *cobra.Command) error {
+	configPath, _ := cmd.Flags().GetString("config")
+	seed, _ := cmd.Flags().GetInt64("seed")
+	forcedModel, _ := cmd.Flags().GetString("model")
+	forcedTLS, _ := cmd.Flags().GetString("tls")
+	forcedLang, _ := cmd.Flags().GetString("lang")
+
+	if seed == 0 {
+		seed = time.Now().UnixNano()
+	}
+	rnd := rand.New(rand.NewSource(seed))
+	profile := core.RandomIdentity(rnd)
+
+	if forcedModel != "" {
+		profile.Model = forcedModel
+		profile.AuthUserAgent = profile.AuthUserAgentFor()
+	}
+	if forcedTLS != "" {
+		profile.TLSFingerprint = forcedTLS
+	}
+	if forcedLang != "" {
+		profile.Language = forcedLang
+	}
+
+	if err := core.SaveProfile(configPath, profile); err != nil {
+		return fmt.Errorf("saving identity profile: %w", err)
+	}
+
+	fmt.Println("Generated a new device identity:")
+	fmt.Println("  " + profile.Describe())
+	fmt.Println("  auth user-agent : " + profile.AuthUserAgent)
+	fmt.Println("  profile file    : " + core.ProfilePath(configPath))
+	fmt.Println("  seed            : " + fmt.Sprint(profile.Seed))
+	fmt.Println()
+	fmt.Println("This profile is used automatically on the next run.")
+	return nil
+}

+// ensureProfileLoaded applies a stored profile to the process defaults.
+func ensureProfileLoaded(configPath string) {
+	if p, err := core.LoadProfile(configPath); err == nil {
+		p.ApplyToDefaults()
+	}
